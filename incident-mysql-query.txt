
-- Create Database
CREATE DATABASE IF NOT EXISTS incidenttrainingdb;
USE incidenttrainingdb;

-- User Table
CREATE TABLE UserTbl (
    UserId INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    FullName VARCHAR(255) NOT NULL,
    UserRole VARCHAR(100) NOT NULL,
    UserEmail VARCHAR(255) UNIQUE NOT NULL,
    UserPassword VARCHAR(255) NOT NULL,
    OrgUserId VARCHAR(100),
    Status VARCHAR(50) NOT NULL DEFAULT 'Active'
);

-- Scenario Table (with ShortSummary)
CREATE TABLE ScenarioTbl (
    ScenarioId VARCHAR(10) PRIMARY KEY,
    ScenarioType VARCHAR(100) NOT NULL,
    Industry VARCHAR(100),
    ShortSummary VARCHAR(255) NOT NULL,  -- For dropdown display
    Summary TEXT,
    WhatWentWrong TEXT,
    SitePhoto TEXT
);

-- Performance Criteria Table
CREATE TABLE PerformanceCriteria (
    PC VARCHAR(10) PRIMARY KEY,
    PCTitle VARCHAR(255) NOT NULL
);

-- Questions Table
CREATE TABLE QuestionsTbl (
    QuestionID VARCHAR(50) PRIMARY KEY,
    PC VARCHAR(10),
    ScenarioID VARCHAR(50),
    QuestionText TEXT,
    Type VARCHAR(50),
    ImageURL VARCHAR(255),
    OptionsJSON JSON,
    CorrectAnswersJSON JSON,
    FormDownloadURL VARCHAR(255),
    FiveWhysJSON JSON,
    HintJSON JSON,
    CreatedDate DATETIME DEFAULT CURRENT_TIMESTAMP,
    UpdatedDate DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- Results Table (with PracticeStatus)
CREATE TABLE ResultsTbl (
    ResultId INT AUTO_INCREMENT PRIMARY KEY,
    StudentId INT UNSIGNED NOT NULL,
    QuestionId VARCHAR(50) NOT NULL,
    ChosenAnswer TEXT,
    UploadedDocLink VARCHAR(255),
    Grade DECIMAL(5,2),
    FacultyComments TEXT,
    PracticeStatus VARCHAR(50) NOT NULL,
    CreatedDate DATETIME DEFAULT CURRENT_TIMESTAMP
);
